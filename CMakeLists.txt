cmake_minimum_required(VERSION 3.16)
project(telemetry_gateway VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(serial_frame STATIC vendor/serial-frame/serial_frame.c)
target_include_directories(serial_frame PUBLIC vendor)
add_library(sqlite3 STATIC vendor/sqlite3.c)
target_include_directories(sqlite3 PUBLIC vendor)
target_compile_definitions(sqlite3 PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION)
target_link_libraries(sqlite3 PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
add_executable(gateway src/main.cpp src/gateway.cpp)
target_include_directories(gateway PRIVATE src vendor)
target_link_libraries(gateway PRIVATE serial_frame sqlite3 Threads::Threads)
if(WIN32)
    target_link_libraries(gateway PRIVATE ws2_32 crypt32)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(gateway PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
find_package(Python3 COMPONENTS Interpreter)
if(BUILD_TESTING AND Python3_Interpreter_FOUND)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_integration.py --binary $<TARGET_FILE:gateway>)
endif()
